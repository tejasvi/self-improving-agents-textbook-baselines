I rewrote four files. The goals are a disciplined solve-and-verify workflow, tools that don't hang or overflow the context, and fixes to bugs in the tool-calling loop. None of this has been run.

**Workflow and prompts (`coding_agent.py`)**
- **Three phases:**
  1. The main solve phase.
  2. A retry with a fresh history if no files were changed.
  3. An independent verification phase that sees the current diff and reviews, re-tests and stress-tests the solution.
- **Detailed solve prompt:** it walks the agent through these steps:
  - locate the solution file
  - analyse constraints and choose the algorithm
  - write the solution
  - test on the samples
  - write a brute force and stress-test against it
  - time the worst case
- **Output rules:** the prompt also covers keeping the `Solution` signature, using only the standard library, reading input fast, avoiding deep recursion, and printing no debug output.
- **Scratch files go to `/tmp`:** untracked files in the repo end up in the final patch, so the agent is told to keep the repo clean.

**Tool loop (`llm_withtools.py`)**
- **Parallel tool calls:** the old loop answered only the first `tool_use` block. A response containing several would break the API call, and the old code swallowed that error silently. Now every block gets a result.
- **Context protection:**
  - Tool outputs are truncated, keeping the head and tail.
  - Old tool results are compacted, except the 8 most recent.
  - There is a cap of 70 tool iterations, since unbounded loops are forbidden.
  - Output tokens are raised to 8192.
- **Error visibility:** tool calls and errors are now logged instead of silently swallowed.
- **o3 bug fix:** `check_for_tool_use` no longer returns a bogus tool call when the response contains no function call.

**bash tool (`tools/bash.py`)**
- Replaced the `while True` reading of private asyncio buffers with `subprocess` and a hard timeout.
- On timeout it kills the whole process group, so leftover child processes can't keep it hanging.
- Stdin is closed, so programs waiting for input don't block.
- Output is truncated, and an optional `timeout` parameter is available.
- The description now states honestly that each call runs in a fresh shell.

**editor tool (`tools/edit.py`)**
- Added `view_range`, `str_replace` and `insert`.
- `view`, `create` and `edit` keep their old behaviour, so the existing tests still apply.

def _truncate(text, limit=MAX_OUTPUT_CHARS):
    if len(text) <= limit:
        return text
    head = limit * 2 // 3
    tail = limit - head
    omitted = len(text) - head - tail
    return text[:head] + f"\n... <{omitted} characters truncated> ...\n" + text[-tail:]


def run_command(command, timeout=None):
    """Run a command in a fresh bash process. Returns (stdout, stderr, returncode, timed_out)."""
    try:
        t = int(timeout) if timeout is not None else DEFAULT_TIMEOUT
    except Exception:
        t = DEFAULT_TIMEOUT
    t = max(1, min(t, MAX_TIMEOUT))

    proc = subprocess.Popen(
        ["/bin/bash", "-c", command],
        stdin=subprocess.DEVNULL,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        start_new_session=True,
        env=os.environ.copy(),
    )
    timed_out = False
    try:
        out, err = proc.communicate(timeout=t)
    except subprocess.TimeoutExpired:
        timed_out = True
        try:
            os.killpg(proc.pid, signal.SIGKILL)
        except Exception:
            try:
                proc.kill()
            except Exception:
                pass
        try:
            out, err = proc.communicate(timeout=10)
        except Exception:
            out, err = b"", b""
    out = (out or b"").decode("utf-8", errors="replace")
    err = (err or b"").decode("utf-8", errors="replace")
    return out, err, proc.returncode, timed_out, t


class BashSession:
    """Compatibility wrapper: each run executes in a fresh bash process."""
    def __init__(self, timeout=DEFAULT_TIMEOUT):
        self._timeout = timeout

    def run(self, command):
        out, err, _, _, _ = run_command(command, self._timeout)
        return out.strip(), err.strip()


def tool_function(command, timeout=None):
    try:
        out, err, code, timed_out, t = run_command(command, timeout)
        result = ""
        out = out.strip()
        err = err.strip()
        if out:
            result += _truncate(out)
        if err:
            result += "\nError:\n" + _truncate(err)
        if timed_out:
            result += f"\nError: command timed out after {t} seconds and was killed."
        elif code not in (0, None) and not err:
            result += f"\n(exit code {code})"
        elif code not in (0, None):
            result += f"\n(exit code {code})"
        return result.strip()
    except Exception as e:
        return f"Error: {str(e)}"


if __name__ == "__main__":
    import sys
    if len(sys.argv) < 2:
        print("Usage: python bash.py '<command>'")
    else:
        print(tool_function(' '.join(sys.argv[1:])))

def maybe_truncate(content: str, max_length: int = MAX_VIEW_CHARS) -> str:
    if len(content) > max_length:
        return content[:max_length] + "\n<response clipped>"
    return content


def validate_path(path: str, command: str) -> Path:
    if not isinstance(path, str) or not path:
        raise ValueError("A valid absolute `path` is required.")
    path_obj = Path(path)
    if not path_obj.is_absolute():
        raise ValueError(f"The path {path} is not an absolute path (must start with '/').")

    if command == "view":
        if not path_obj.exists():
            raise ValueError(f"The path {path} does not exist.")
    elif command == "create":
        if path_obj.exists():
            raise ValueError(f"Cannot create new file; {path} already exists. Use `edit` to overwrite it.")
    elif command in ("edit", "str_replace", "insert"):
        if not path_obj.exists():
            raise ValueError(f"The file {path} does not exist.")
        if path_obj.is_dir():
            raise ValueError(f"{path} is a directory and cannot be edited as a file.")
    else:
        raise ValueError(f"Unknown or unsupported command: {command}")
    return path_obj


def format_output(content: str, path: str, init_line: int = 1) -> str:
    content = maybe_truncate(content)
    content = content.expandtabs()
    numbered_lines = [
        f"{i + init_line:6}\t{line}"
        for i, line in enumerate(content.split("\n"))
    ]
    return f"Here's the result of running `cat -n` on {path}:\n" + "\n".join(numbered_lines) + "\n"


def read_file(path: Path) -> str:
    try:
        return path.read_text()
    except Exception as e:
        raise ValueError(f"Failed to read file: {e}")


def write_file(path: Path, content: str):
    try:
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text(content)
    except Exception as e:
        raise ValueError(f"Failed to write file: {e}")


def view_path(path_obj: Path, view_range=None) -> str:
    if path_obj.is_dir():
        try:
            result = subprocess.run(
                ['find', str(path_obj), '-maxdepth', '2', '-not', '-path', '*/\\.*'],
                capture_output=True,
                text=True
            )
            if result.stderr:
                return f"Error listing directory: {result.stderr}"
            return (
                f"Here's the files and directories up to 2 levels deep in {path_obj}, excluding hidden items:\n"
                + maybe_truncate(result.stdout)
            )
        except Exception as e:
            raise ValueError(f"Failed to list directory: {e}")

    content = read_file(path_obj)
    if view_range:
        if not isinstance(view_range, (list, tuple)) or len(view_range) != 2:
            raise ValueError("`view_range` must be a list of two integers [start, end].")
        lines = content.split("\n")
        start, end = int(view_range[0]), int(view_range[1])
        if start < 1:
            start = 1
        if end == -1 or end > len(lines):
            end = len(lines)
        if start > end:
            raise ValueError(f"Invalid view_range {view_range}; file has {len(lines)} lines.")
        return format_output("\n".join(lines[start - 1:end]), str(path_obj), init_line=start)
    return format_output(content, str(path_obj))


def _snippet(content: str, center_line: int, path: str, radius: int = 4) -> str:
    lines = content.split("\n")
    start = max(1, center_line - radius)
    end = min(len(lines), center_line + radius)
    return format_output("\n".join(lines[start - 1:end]), path, init_line=start)


def tool_function(command: str, path: str, file_text: str = None, view_range=None,
                  old_str: str = None, new_str: str = None, insert_line: int = None) -> str:
    try:
        path_obj = validate_path(path, command)

        if command == "view":
            return view_path(path_obj, view_range)

        elif command == "create":
            if file_text is None:
                raise ValueError("Missing required `file_text` for 'create' command.")
            write_file(path_obj, file_text)
            return f"File created successfully at: {path}"

        elif command == "edit":
            if file_text is None:
                raise ValueError("Missing required `file_text` for 'edit' command.")
            write_file(path_obj, file_text)
            return f"File at {path} has been overwritten with new content."

        elif command == "str_replace":
            if old_str is None or old_str == "":
                raise ValueError("Missing required non-empty `old_str` for 'str_replace' command.")
            if new_str is None:
                new_str = ""
            content = read_file(path_obj)
            count = content.count(old_str)
            if count == 0:
                raise ValueError(f"`old_str` was not found verbatim in {path}. View the file and copy the exact text (whitespace matters).")
            if count > 1:
                raise ValueError(f"`old_str` occurs {count} times in {path}; include more context to make it unique.")
            idx = content.index(old_str)
            new_content = content.replace(old_str, new_str, 1)
            write_file(path_obj, new_content)
            line_no = content[:idx].count("\n") + 1
            return f"The file {path} has been edited. " + _snippet(new_content, line_no, path)

        elif command == "insert":
            if new_str is None:
                raise ValueError("Missing required `new_str` for 'insert' command.")
            if insert_line is None:
                raise ValueError("Missing required `insert_line` for 'insert' command.")
            content = read_file(path_obj)
            lines = content.split("\n")
            n = int(insert_line)
            if n < 0 or n > len(lines):
                raise ValueError(f"`insert_line` {n} out of range [0, {len(lines)}].")
            new_lines = lines[:n] + new_str.split("\n") + lines[n:]
            new_content = "\n".join(new_lines)
            write_file(path_obj, new_content)
            return f"The file {path} has been edited. " + _snippet(new_content, n + 1, path)

        else:
            raise ValueError(f"Unknown command: {command}")

    except Exception as e:
        return f"Error: {str(e)}"


if __name__ == "__main__":
    print(tool_function("view", str(Path("./coding_agent.py").resolve()), view_range=[1, 10]))

import ast
import json
import re
import anthropic
import backoff
import openai
import copy

from llm import create_client, get_response_from_llm
from prompts.tooluse_prompt import get_tooluse_prompt
from tools import load_all_tools

CLAUDE_MODEL = 'claude-3-5-sonnet-20241022'
OPENAI_MODEL = 'claude-3-5-sonnet-20241022'

MAX_TOOL_ITERATIONS = 70
MAX_TOOL_RESULT_CHARS = 15000
KEEP_RECENT_TOOL_RESULTS = 8
COMPACT_TOOL_RESULT_CHARS = 1200


def _truncate_text(text, limit=MAX_TOOL_RESULT_CHARS):
    text = str(text)
    if len(text) <= limit:
        return text
    head = limit * 2 // 3
    tail = limit - head
    return text[:head] + f"\n... <{len(text) - head - tail} characters truncated> ...\n" + text[-tail:]


def process_tool_call(tools_dict, tool_name, tool_input):
    try:
        if tool_name in tools_dict:
            if not isinstance(tool_input, dict):
                return f"Error: tool input for '{tool_name}' must be an object."
            return tools_dict[tool_name]['function'](**tool_input)
        else:
            return f"Error: Tool '{tool_name}' not found"
    except Exception as e:
        return f"Error executing tool '{tool_name}': {str(e)}"


@backoff.on_exception(
    backoff.expo,
    (openai.RateLimitError, openai.APITimeoutError, anthropic.RateLimitError, anthropic.APIStatusError),
    max_time=600,
    max_value=60,
)
def get_response_withtools(
    client, model, messages, tools, tool_choice,
    logging=None, max_retry=3
):
    try:
        if 'claude' in model:
            response = client.messages.create(
                model=model,
                messages=messages,
                max_tokens=8192,
                tool_choice=tool_choice,
                tools=tools,
            )
        elif model.startswith('o3-'):
            response = client.responses.create(
                model=model,
                input=messages,
                tool_choice=tool_choice,
                tools=tools,
                parallel_tool_calls=False
            )
        else:
            raise ValueError(f"Unsupported model: {model}")
        return response
    except Exception as e:
        if logging:
            logging(f"Error in get_response_withtools: {str(e)}")
        if 'Input is too long' in str(e) or 'prompt is too long' in str(e):
            raise
        if max_retry > 0:
            return get_response_withtools(client, model, messages, tools, tool_choice, logging, max_retry - 1)
        raise


def check_for_tool_use(response, model=''):
    """
    Checks if the response contains a tool call.
    """
    if 'claude' in model:
        if response.stop_reason == "tool_use":
            tool_use_block = next((block for block in response.content if block.type == "tool_use"), None)
            if tool_use_block is not None:
                return {
                    'tool_id': tool_use_block.id,
                    'tool_name': tool_use_block.name,
                    'tool_input': tool_use_block.input,
                }

    elif model.startswith('o3-'):
        tool_call = None
        for item in response.output:
            if item.type == "function_call":
                tool_call = item
                break
        if tool_call is not None:
            return {
                'tool_id': tool_call.call_id,
                'tool_name': tool_call.name,
                'tool_input': json.loads(tool_call.arguments),
            }

    else:
        pattern = r'<tool_use>(.*?)</tool_use>'
        match = re.search(pattern, response, re.DOTALL)
        if match:
            tool_use_str = match.group(1).strip()
            try:
                tool_use_dict = ast.literal_eval(tool_use_str)
                if isinstance(tool_use_dict, dict) and 'tool_name' in tool_use_dict and 'tool_input' in tool_use_dict:
                    return tool_use_dict
            except Exception:
                pass

    return None


def convert_tool_info(tool_info, model=None):
    """
    Converts tool_info from Claude format to the given model's format.
    """
    if 'claude' in model:
        return {
            'name': tool_info['name'],
            'description': tool_info['description'],
            'input_schema': tool_info['input_schema'],
        }
    elif model.startswith('o3-'):
        def add_additional_properties(d):
            if isinstance(d, dict):
                if 'properties' in d:
                    d['additionalProperties'] = False
                for k, v in d.items():
                    add_additional_properties(v)
        add_additional_properties(tool_info['input_schema'])
        for p in tool_info['input_schema']['properties'].keys():
            if not p in tool_info['input_schema']['required']:
                tool_info['input_schema']['required'].append(p)
                t = copy.deepcopy(tool_info['input_schema']['properties'][p]["type"])
                if isinstance(t, str):
                    tool_info['input_schema']['properties'][p]["type"] = [t, "null"]
                elif isinstance(t, list):
                    tool_info['input_schema']['properties'][p]["type"] = t + ["null"]

        return {
            'type': 'function',
            'name': tool_info['name'],
            'description': tool_info['description'],
            'parameters': tool_info['input_schema'],
            "strict": True,
        }
    else:
        return tool_info


def convert_block_claude(block):
    if isinstance(block, dict):
        block_type = block.get('type')
        text = block.get('text')
        tool_name = block.get('name')
        tool_input = block.get('input')
        tool_result = block.get('content')
    else:
        block_type = getattr(block, 'type', None)
        text = getattr(block, 'text', None)
        tool_name = getattr(block, 'name', None)
        tool_input = getattr(block, 'input', None)
        tool_result = getattr(block, 'content', None)

    text = text or ""

    if block_type == "text":
        return {"type": "text", "text": text}
    elif block_type == "tool_use":
        return {
            "type": "text",
            "text": f"<tool_use>\n{{'tool_name': {tool_name}, 'tool_input': {tool_input}}}\n</tool_use>"
        }
    elif block_type == "tool_result":
        return {"type": "text", "text": f"Tool Result: {tool_result}"}
    else:
        return {"type": "text", "text": str(block)}


def convert_msg_history_claude(msg_history):
    new_msg_history = []
    for msg in msg_history:
        role = msg.get('role', '')
        content_blocks = msg.get('content', [])
        if isinstance(content_blocks, str):
            content_blocks = [{"type": "text", "text": content_blocks}]
        new_content = [convert_block_claude(block) for block in content_blocks]
        new_msg_history.append({"role": role, "content": new_content})
    return new_msg_history


def convert_msg_history_openai(msg_history):
    new_msg_history = []
    for msg in msg_history:
        if isinstance(msg, dict):
            role = msg.get('role', '')
            content = msg.get('content', '')
            if role == 'tool':
                new_msg = {
                    "role": "user",
                    "content": [{"type": "text", "text": f"Tool Result: {content}"}],
                }
            else:
                new_msg = {"role": role, "content": content}
        else:
            role = getattr(msg, 'role', None)
            content = getattr(msg, 'content', None)
            tool_calls = getattr(msg, 'tool_calls', None)
            if tool_calls:
                tool_call = tool_calls[0]
                function_name = getattr(tool_call.function, 'name', '')
                function_args = getattr(tool_call.function, 'arguments', '')
                new_msg = {
                    "role": role,
                    "content": [{
                        "type": "text",
                        "text": f"<tool_use>\n{{'tool_name': {function_name}, 'tool_input': {function_args}}}\n</tool_use>",
                    }],
                }
            else:
                new_msg = {"role": role, "content": [{"type": "text", "text": content}]}
        new_msg_history.append(new_msg)
    return new_msg_history


def convert_msg_history(msg_history, model=None):
    if 'claude' in model:
        return convert_msg_history_claude(msg_history)
    elif model.startswith('o3-'):
        return convert_msg_history_openai(msg_history)
    else:
        return msg_history


def chat_with_agent_manualtools(msg, model, msg_history=None, logging=print):
    if msg_history is None:
        msg_history = []
    system_message = f'You are a coding agent.\n\n{get_tooluse_prompt()}'
    new_msg_history = msg_history

    try:
        all_tools = load_all_tools(logging=logging)
        tools_dict = {tool['info']['name']: tool for tool in all_tools}
        client, client_model = create_client(model)

        logging(f"Input: {msg}")
        response, new_msg_history = get_response_from_llm(
            msg=msg,
            client=client,
            model=client_model,
            system_message=system_message,
            print_debug=False,
            msg_history=new_msg_history,
        )
        logging(f"Output: {response}")

        tool_use = check_for_tool_use(response, model=client_model)
        for _ in range(MAX_TOOL_ITERATIONS):
            if not tool_use:
                break
            tool_name = tool_use['tool_name']
            tool_input = tool_use['tool_input']
            tool_result = _truncate_text(process_tool_call(tools_dict, tool_name, tool_input))

            tool_msg = f'Tool Used: {tool_name}\nTool Input: {tool_input}\nTool Result: {tool_result}'
            logging(tool_msg)
            response, new_msg_history = get_response_from_llm(
                msg=tool_msg,
                client=client,
                model=client_model,
                system_message=system_message,
                print_debug=False,
                msg_history=new_msg_history,
            )
            logging(f"Output: {response}")
            tool_use = check_for_tool_use(response, model=client_model)

    except Exception as e:
        logging(f"Error in chat_with_agent_manualtools: {e}")

    return new_msg_history


def _compact_old_tool_results(history):
    """Shrink older tool results to keep the context manageable."""
    tool_result_blocks = []
    for m in history:
        if m.get("role") == "user" and isinstance(m.get("content"), list):
            for b in m["content"]:
                if isinstance(b, dict) and b.get("type") == "tool_result":
                    tool_result_blocks.append(b)
    if len(tool_result_blocks) <= KEEP_RECENT_TOOL_RESULTS:
        return
    for b in tool_result_blocks[:-KEEP_RECENT_TOOL_RESULTS]:
        c = b.get("content")
        if isinstance(c, str) and len(c) > COMPACT_TOOL_RESULT_CHARS:
            keep = COMPACT_TOOL_RESULT_CHARS // 2
            b["content"] = c[:keep] + "\n... <old tool output compacted> ...\n" + c[-keep:]


def chat_with_agent_claude(
        msg,
        model='bedrock/us.anthropic.claude-3-5-sonnet-20241022-v2:0',
        msg_history=None,
        logging=print,
    ):
    if msg_history is None:
        msg_history = []
    new_msg_history = [
        {"role": "user", "content": [{"type": "text", "text": msg}]}
    ]
    separator = '=' * 10
    logging(f"\n{separator} User Instruction {separator}\n{msg}")

    try:
        client, client_model = create_client(model)
        all_tools = load_all_tools(logging=logging)
        tools_dict = {tool['info']['name']: tool for tool in all_tools}
        tools = [convert_tool_info(tool['info'], model=client_model) for tool in all_tools]

        response = get_response_withtools(
            client=client,
            model=client_model,
            messages=msg_history + new_msg_history,
            tool_choice={"type": "auto"},
            tools=tools,
            logging=logging,
        )

        for iteration in range(MAX_TOOL_ITERATIONS):
            for block in response.content:
                if getattr(block, "type", None) == "text" and getattr(block, "text", None):
                    logging(f"\n{separator} Agent {separator}\n{block.text}")

            tool_uses = [b for b in response.content if getattr(b, "type", None) == "tool_use"]
            if response.stop_reason != "tool_use" or not tool_uses:
                break

            new_msg_history.append({"role": "assistant", "content": response.content})
            results = []
            for tu in tool_uses:
                tool_result = process_tool_call(tools_dict, tu.name, tu.input)
                tool_result = _truncate_text(tool_result if tool_result is not None else "")
                if not tool_result.strip():
                    tool_result = "(no output)"
                logging(f"Tool Used: {tu.name}\nTool Input: {tu.input}\nTool Result: {tool_result}")
                results.append({
                    "type": "tool_result",
                    "tool_use_id": tu.id,
                    "content": tool_result,
                })
            new_msg_history.append({"role": "user", "content": results})

            if iteration == MAX_TOOL_ITERATIONS - 2:
                new_msg_history[-1]["content"].append({
                    "type": "text",
                    "text": "Tool budget almost exhausted. Make sure the final solution is saved in the repository now.",
                })

            _compact_old_tool_results(new_msg_history)

            response = get_response_withtools(
                client=client,
                model=client_model,
                messages=msg_history + new_msg_history,
                tool_choice={"type": "auto"},
                tools=tools,
                logging=logging,
            )

        if response.stop_reason == "tool_use":
            # Iteration cap reached with a pending tool call; do not leave a dangling tool_use.
            final_response = "(Stopped: tool iteration limit reached.)"
        else:
            final_response = next((block.text for block in response.content if hasattr(block, "text")), None) or ""
        new_msg_history.append({
            "role": "assistant",
            "content": [{"type": "text", "text": final_response or "(no text)"}],
        })

    except Exception as e:
        logging(f"Error in chat_with_agent_claude: {e}")

    return new_msg_history

        logging=print,

        )

        for _ in range(MAX_TOOL_ITERATIONS):
            if not tool_use:
                break
            tool_name = tool_use['tool_name']
            tool_input = tool_use['tool_input']
            if isinstance(tool_input, dict):
                tool_input = {k: v for k, v in tool_input.items() if v is not None}
            tool_result = _truncate_text(process_tool_call(tools_dict, tool_name, tool_input))

            logging(f"Tool Used: {tool_name}")
            logging(f"Tool Input: {tool_input}")
            logging(f"Tool Result: {tool_result}")

            tool_call = None
            for item in response.output:
                if item.type == "function_call":
                    tool_call = item
                    break
            new_msg_history.append(tool_call)
            new_msg_history.append({
                "type": "function_call_output",
                "call_id": tool_use['tool_id'],
                "output": tool_result,
            })
            response = get_response_withtools(
                client=client,
                model=client_model,
                messages=msg_history + new_msg_history,
                tool_choice="auto",
                tools=tools,
                logging=logging,
            )
            tool_use = check_for_tool_use(response, model=client_model)
            logging(f"Tool Response: {response}")

        new_msg_history.append(response)

    except Exception as e:
        logging(f"Error in chat_with_agent_openai: {e}")

    return new_msg_history


def chat_with_agent(
    msg,
    model=CLAUDE_MODEL,
    msg_history=None,
    logging=print,
    convert=False,
):
    if msg_history is None:
        msg_history = []

    if 'claude' in model:
        new_msg_history = chat_with_agent_claude(msg, model=model, msg_history=msg_history, logging=logging)
        conv_msg_history = convert_msg_history(new_msg_history, model=model)
        if convert:
            new_msg_history = conv_msg_history
        new_msg_history = msg_history + new_msg_history

    elif model.startswith('o3-'):
        new_msg_history = chat_with_agent_openai(msg, model=model, msg_history=msg_history, logging=logging)
        new_msg_history = msg_history + new_msg_history

    else:
        new_msg_history = chat_with_agent_manualtools(msg, model=model, msg_history=msg_history, logging=logging)
        conv_msg_history = convert_msg_history(new_msg_history, model=model)
        if convert:
            new_msg_history = conv_msg_history

    return new_msg_history
